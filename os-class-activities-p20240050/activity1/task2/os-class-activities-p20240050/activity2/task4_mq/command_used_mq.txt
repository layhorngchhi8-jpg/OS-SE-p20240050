    mq_getattr(mq, &attr);

    if (mq_receive(mq, buffer, attr.mq_msgsize, NULL) == -1) { perror("mq_receive"); exit(EXIT_FAILURE); }

    printf("Receiver: message = \"%s\"\n", buffer);

    printf("Receiver: queue unlinked.\n");
